*  SYNOPSIS
Convert a publicly available listing of m3u8 urls to live bbc radio
streams to an org-file with audio links which can be opened by
org-media-note.

#+begin_src emacs-lisp :results silent
;;  -*- lexical-binding: t -*-
(cl-defstruct (bbc-elt (:type list)) caption uri frag)

(cl-defun parse-bbc-uri (uri &key caption bitrate force-https)
  (when (string-match
	 "^http\\(s\\)?://.*/\\([^/]*\\)-audio%3d\\([0-9]+\\)\\.norewind\\.m3u8$"
	 uri)
    (let ((tlsp (match-string 1 uri))
	  (frag (match-string 2 uri))
	  (rate (match-string 3 uri)))
      (when (and bitrate
		 (not (equal bitrate rate)))
	(setq uri (replace-match bitrate nil t uri 3)))
      (when (and force-https
		 (not tlsp)
		 (string-match "^\\(http://\\)" uri))
	(setq uri (replace-match "https://" nil t uri 1)))
      (make-bbc-elt :caption caption :uri uri :frag frag))))

(cl-assert (cl-equalp
	    (parse-bbc-uri "http://host/frag-audio%3d320000.norewind.m3u8"
			   :bitrate "42000"
			   :force-https t)
	    (make-bbc-elt :uri "https://host/frag-audio%3d42000.norewind.m3u8"
			  :frag "frag")))

(cl-defun read-bbc-gist (file &rest args &key (bitrate "48000") (force-https t))
  (when bitrate
    (cl-assert (cl-find bitrate  '("320000" "96000" "48000") :test #'equal)))
  (cl-flet ((slurp-lines (file)
	      (with-temp-buffer
		(insert-file-contents file)
		(split-string (buffer-string) "[\n]" t "[ \f\t\n\r\v]+"))))
    (cl-loop for (header uri . rest)  on (cdr (slurp-lines file)) by #'cddr
	     for caption = (and header
				(string-match "^#EXTINF:-1,BBC - \\(.*\\)$"
					      header)
				(match-string 1 header))
	     when (and header caption
		       (parse-bbc-uri uri :caption caption :bitrate bitrate
				      :force-https force-https))
	     collect it)))

(defun bbc-elt-sched-url (elt)
  (let ((frag (bbc-elt-frag elt)))
    (format "https://www.bbc.co.uk/sounds/schedules/%s" frag)))

(defun bbc-elt-live-url (elt)
  (let ((frag (bbc-elt-frag elt)))
    (format "https://www.bbc.co.uk/sounds/play/live/%s" frag)))

(cl-defun dump-1org (elts file)
  (with-temp-buffer
    (cl-loop for elt in elts
	     with sep = "\n"
	     do
	     (insert (concat "** " (bbc-elt-caption elt) sep
			     (format "[[audio:%s][%s]]" (bbc-elt-uri elt)
				     (bbc-elt-frag elt))
			     sep
			     (format "[[%s][sched]]" (bbc-elt-sched-url elt))
			     sep
			     (format "[[%s][live]]" (bbc-elt-live-url elt))
			     sep)))
    (write-file file)))

(defun dump-2org (elts file)
  (with-temp-buffer
    (cl-loop for elt in elts
	     with sep = " | " and sep-b = "| " and sep-e = " |\n"
	     do
	     (insert (concat sep-b (bbc-elt-caption elt) sep
			     (format "[[audio:%s][%s]]" (bbc-elt-uri elt)
				     (bbc-elt-frag elt))
			     sep
			     (format "[[%s][sched]]" (bbc-elt-sched-url elt))
			     sep
			     (format "[[%s][live]]" (bbc-elt-live-url elt))
			     sep-e)))
    (write-file file)))

(defun org-to-bare-html (in-file out-file &optional async subtreep visible-only body-only ext-plist)
  (with-temp-buffer
    (insert-file-contents in-file)
    (let ((org-html-head "")
	  (org-html-head-extra "")
	  (org-html-head-include-default-style nil)
	  (org-html-head-include-scripts nil)
	  (org-html-preamble nil)
	  (org-html-postamble nil)
	  (org-html-use-infojs nil)
	  (org-export-with-sub-superscripts nil))
      (let* ((extension (concat
			 (when (> (length org-html-extension) 0) ".")
			 (or (plist-get ext-plist :html-extension)
			     org-html-extension
			     "html")))
	     (file (or out-file (org-export-output-file-name extension subtreep)))
	     (org-export-coding-system org-html-coding-system))
	(org-export-to-file 'html file
	  async subtreep visible-only body-only ext-plist)))))

* Rewind
Upto 6, possibly 24 hours.

#+begin_src elisp :eval no
;; "The media segment names are just epoch / 6.4 so you can construct
;; a playlist for any time and date. Presumably the official
;; rewindable playlist has 6 hours of segments because that is their
;; guaranteed retention. That said, I've tested quickly and can play
;; back segments from at least 12 hours ago."
;;
;;  -- johnpenny (2026-07-15) gist permalink_comment_id=6256527

(defun seg0 (timestamp)
  (floor timestamp 6.4))

(defun uri-prefix-for-frag (frag elts)
  " ELTS is an object returned by calling `read-bbc-gist'."
  ;; uri = (cl-rassoc $rfrag $elts :test #'equal :key #'cadr)
  (let ((uri (cl-loop for (_caption uri frag1) in elts
		      when (equal frag frag1)
		      return uri)))
    (let ((p (cl-position ?\/ uri :from-end t)))
      (cl-subseq uri 0 (1+ p)))))

(cl-defun dump-m3u8 (&key file seg0 (count 5) frag (bitrate 48000)
			  uri-prefix)
  "Write a m3u8 FILE with COUNT segments starting at segment number SEG0
for the BBC station FRAG at the given BITRATE.  Each segment is
probably lasts 6 seconds.  If FILE is NIL return the contents as a
string."
  (let* ((bitrate (if (stringp bitrate) bitrate (number-to-string bitrate)))
	 (body (with-temp-buffer
		 (insert "#EXTM3U
#EXT-X-VERSION:3
#EXT-X-MEDIA-SEQUENCE:"
			 (number-to-string seg0)
			 "\n#EXT-X-INDEPENDENT-SEGMENTS
#EXT-X-TARGETDURATION:6\n\n")
		 (cl-loop for i below count
			  for segment from seg0
			  do (insert "#EXTINF:6.4, no desc\n"
				     uri-prefix frag "-audio="
				     bitrate "-" (number-to-string segment)
				     ".ts\n"))
		 (insert "#EXT-X-ENDLIST\n")
		 (buffer-string))))
    (if file
	(with-temp-buffer
	  (insert body)
	  (write-file file))
      body)))
